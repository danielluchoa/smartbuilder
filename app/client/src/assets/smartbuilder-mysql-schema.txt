-- ============================================================================
-- SmartBuilder — Production MySQL schema (database-per-tenant)
-- Version: 1.0 — 2026-10-04
-- ============================================================================
--
-- DESIGN
--   1. One PLATFORM database:  smartbuilder_platform
--        Registry of tenants + platform owner logins (Daniel's admin area).
--   2. One DATABASE PER TENANT: smartbuilder_tenant_<tenant_id>
--        Each customer company gets its own MySQL database. No customer
--        ever shares tables with another customer.
--
-- MIGRATION FROM THE PILOT
--   The pilot stores every row with a company_id (tenant key). To migrate
--   a tenant to production:
--     1. In the owner panel: Tenants -> Export tenant database (SQL).
--        The export contains INSERTs already stripped of company_id.
--     2. CREATE DATABASE smartbuilder_tenant_<id>; run this schema file.
--     3. Import the tenant's SQL dump.
--     4. Register the tenant row in smartbuilder_platform.tenants.
--
-- CONVENTIONS
--   Engine=InnoDB, utf8mb4. Money = DECIMAL(12,2) USD. Time = DATETIME
--   stored in the tenant's local timezone (company setting). Coordinates
--   = DECIMAL(10,7). Distances in feet.
--
-- ============================================================================
-- PART 1 — PLATFORM DATABASE
-- ============================================================================

CREATE DATABASE IF NOT EXISTS smartbuilder_platform
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE smartbuilder_platform;

-- Platform owners (Daniel's super-admin logins)
CREATE TABLE platform_owners (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name          VARCHAR(120) NOT NULL,
  email         VARCHAR(190) NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_owner_email (email)
) ENGINE=InnoDB;

-- Tenant registry: one row per customer company -> one database
CREATE TABLE tenants (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  company_name VARCHAR(160) NOT NULL,
  db_name      VARCHAR(64)  NOT NULL COMMENT 'e.g. smartbuilder_tenant_12',
  db_host      VARCHAR(120) NOT NULL DEFAULT 'localhost',
  status       ENUM('trial','active','suspended') NOT NULL DEFAULT 'trial',
  plan         VARCHAR(60)  NOT NULL DEFAULT 'standard',
  created_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_tenant_db (db_name)
) ENGINE=InnoDB;

-- ============================================================================
-- PART 2 — TENANT DATABASE (run once per tenant)

-- ---- Company profile -------------------------------------------------------
CREATE TABLE companies (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name       VARCHAR(160) NOT NULL,
  timezone   VARCHAR(64)  NOT NULL DEFAULT 'America/New_York',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id)
) ENGINE=InnoDB;

CREATE TABLE company_settings (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  setting_key VARCHAR(80)  NOT NULL,
  setting_value VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY uq_setting (setting_key)
) ENGINE=InnoDB COMMENT 'auto_checkout_time=17:00, no_show_cutoff=09:00, geofence_default_ft=656, ot_enabled=0, ot_daily_hours=8, ot_weekly_hours=40, ot_multiplier=1.5, ...';

-- ---- Users -----------------------------------------------------------------
CREATE TABLE users (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name          VARCHAR(120) NOT NULL,
  email         VARCHAR(190) NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  role          ENUM('admin','manager','employee') NOT NULL DEFAULT 'employee',
  status        ENUM('active','inactive') NOT NULL DEFAULT 'active',
  phone         VARCHAR(40)  NULL,
  pay_type      ENUM('hourly','daily','contract') NOT NULL DEFAULT 'hourly',
  pay_rate      DECIMAL(12,2) NOT NULL DEFAULT 0 COMMENT 'per hour, per day, or contract amount',
  created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_user_email (email),
  KEY idx_user_role (role)
) ENGINE=InnoDB;

-- ---- Clients (1 client -> N projects, or none) ------------------------------
CREATE TABLE clients (
  id             BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name           VARCHAR(160) NOT NULL,
  contact_person VARCHAR(120) NULL,
  phone          VARCHAR(40)  NULL,
  email          VARCHAR(190) NULL,
  bill_number    VARCHAR(20)  NULL,
  bill_street    VARCHAR(160) NULL,
  bill_city      VARCHAR(100) NULL,
  bill_state     CHAR(2)      NULL,
  bill_zip       VARCHAR(12)  NULL,
  created_at     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_client_name (name)
) ENGINE=InnoDB;

-- ---- Projects (job sites) ---------------------------------------------------
CREATE TABLE projects (
  id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  client_id       BIGINT UNSIGNED NULL COMMENT 'optional; a project may have no client',
  name            VARCHAR(160) NOT NULL,
  scope_of_work   TEXT NULL COMMENT 'shown to crew as "What to do"',
  status          ENUM('scheduled','in_progress','done') NOT NULL DEFAULT 'scheduled',
  site_number     VARCHAR(20)  NULL,
  site_street     VARCHAR(160) NULL,
  site_city       VARCHAR(100) NULL,
  site_state      CHAR(2)      NULL,
  site_zip        VARCHAR(12)  NULL,
  geofence_lat    DECIMAL(10,7) NULL COMMENT 'auto-filled by address geocoding',
  geofence_lng    DECIMAL(10,7) NULL,
  geofence_radius_ft INT UNSIGNED NOT NULL DEFAULT 656,
  estimated_value DECIMAL(12,2) NULL COMMENT 'contract/quoted price',
  start_date      DATE NULL,
  end_date        DATE NULL,
  created_at      DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_project_client (client_id),
  KEY idx_project_status (status),
  CONSTRAINT fk_project_client FOREIGN KEY (client_id)
    REFERENCES clients (id) ON DELETE SET NULL
) ENGINE=InnoDB;

-- ---- Jobs (trades inside a project: Roof, Siding... different crews) --------
CREATE TABLE jobs (
  id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  project_id      BIGINT UNSIGNED NOT NULL,
  name            VARCHAR(120) NOT NULL COMMENT 'e.g. Roof replacement',
  scope           TEXT NULL COMMENT 'what the crew must do',
  status          ENUM('scheduled','in_progress','done') NOT NULL DEFAULT 'scheduled',
  estimated_value DECIMAL(12,2) NULL,
  start_date      DATE NULL,
  end_date        DATE NULL,
  created_at      DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_job_project (project_id),
  CONSTRAINT fk_job_project FOREIGN KEY (project_id)
    REFERENCES projects (id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---- Job tasks (checklist inside each job, worked by the crew) --------------
-- assignee_id NULL = whole job crew. In Progress persists across days until
-- the worker marks Done; started_at/completed_at drive elapsed-time display.
CREATE TABLE job_tasks (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  job_id       BIGINT UNSIGNED NOT NULL,
  title        VARCHAR(191) NOT NULL,
  notes        TEXT NULL,
  assignee_id  BIGINT UNSIGNED NULL COMMENT 'NULL = whole job crew',
  status       ENUM('todo','in_progress','done') NOT NULL DEFAULT 'todo',
  sort_order   INT NOT NULL DEFAULT 0,
  created_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  started_at   DATETIME NULL,
  completed_at DATETIME NULL,
  PRIMARY KEY (id),
  KEY idx_task_job (job_id, status, sort_order),
  CONSTRAINT fk_task_job      FOREIGN KEY (job_id)      REFERENCES jobs (id)  ON DELETE CASCADE,
  CONSTRAINT fk_task_assignee FOREIGN KEY (assignee_id) REFERENCES users (id) ON DELETE SET NULL
) ENGINE=InnoDB;

-- ---- Task photos (Before / During / After evidence per task) ----------------
CREATE TABLE task_photos (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  task_id    BIGINT UNSIGNED NOT NULL,
  user_id    BIGINT UNSIGNED NOT NULL COMMENT 'who uploaded it',
  photo_url  LONGTEXT NULL COMMENT 'inline device image (data:image/...)',
  stage      ENUM('before','during','after') NOT NULL DEFAULT 'during',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_photo_task (task_id, created_at),
  CONSTRAINT fk_photo_task FOREIGN KEY (task_id) REFERENCES job_tasks (id) ON DELETE CASCADE,
  CONSTRAINT fk_photo_user FOREIGN KEY (user_id) REFERENCES users (id)
) ENGINE=InnoDB;

-- ---- Vendors ----------------------------------------------------------------
CREATE TABLE vendors (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name       VARCHAR(160) NOT NULL,
  contact    VARCHAR(120) NULL,
  phone      VARCHAR(40)  NULL,
  email      VARCHAR(190) NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id)
) ENGINE=InnoDB;

-- ---- Crew assignments (who works which job/project, per day) ---------------
CREATE TABLE assignments (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id    BIGINT UNSIGNED NOT NULL,
  project_id BIGINT UNSIGNED NOT NULL,
  job_id     BIGINT UNSIGNED NULL COMMENT 'NULL = whole project',
  work_date  DATE NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_assign_user_date (user_id, work_date),
  KEY idx_assign_project (project_id),
  CONSTRAINT fk_assign_user    FOREIGN KEY (user_id)    REFERENCES users (id)    ON DELETE CASCADE,
  CONSTRAINT fk_assign_project FOREIGN KEY (project_id) REFERENCES projects (id) ON DELETE CASCADE,
  CONSTRAINT fk_assign_job     FOREIGN KEY (job_id)     REFERENCES jobs (id)     ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---- Timesheets (anti-fraud: GPS-validated, tamper-proof) -------------------
CREATE TABLE timesheets (
  id                 BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id            BIGINT UNSIGNED NOT NULL,
  project_id         BIGINT UNSIGNED NOT NULL,
  job_id             BIGINT UNSIGNED NULL,
  check_in           DATETIME NOT NULL,
  check_out          DATETIME NULL,
  check_in_lat       DECIMAL(10,7) NULL,
  check_in_lng       DECIMAL(10,7) NULL,
  check_out_lat      DECIMAL(10,7) NULL,
  check_out_lng      DECIMAL(10,7) NULL,
  check_in_inside    TINYINT(1) NULL COMMENT '1=inside geofence, 0=flagged OUTSIDE',
  check_out_inside   TINYINT(1) NULL,
  hours_worked       DECIMAL(6,2) NULL COMMENT 'computed at check-out/auto-close',
  pay_type_snapshot  ENUM('hourly','daily','contract') NOT NULL COMMENT 'frozen at check-in; history never rewrites',
  pay_rate_snapshot  DECIMAL(12,2) NOT NULL,
  labor_cost         DECIMAL(12,2) NULL COMMENT 'computed from snapshot at approval',
  approval_status    ENUM('pending','approved','rejected') NOT NULL DEFAULT 'pending',
  approved_by        BIGINT UNSIGNED NULL,
  auto_closed        TINYINT(1) NOT NULL DEFAULT 0 COMMENT '1=auto check-out at 17:00',
  created_at         DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_ts_user_date (user_id, check_in),
  KEY idx_ts_project (project_id),
  KEY idx_ts_approval (approval_status),
  CONSTRAINT fk_ts_user    FOREIGN KEY (user_id)    REFERENCES users (id),
  CONSTRAINT fk_ts_project FOREIGN KEY (project_id) REFERENCES projects (id),
  CONSTRAINT fk_ts_job     FOREIGN KEY (job_id)     REFERENCES jobs (id) ON DELETE SET NULL,
  CONSTRAINT fk_ts_approver FOREIGN KEY (approved_by) REFERENCES users (id) ON DELETE SET NULL
) ENGINE=InnoDB;

-- GPS trail while the app is open (30-min pings; NOT background tracking)
CREATE TABLE location_pings (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  timesheet_id BIGINT UNSIGNED NOT NULL,
  lat          DECIMAL(10,7) NOT NULL,
  lng          DECIMAL(10,7) NOT NULL,
  recorded_at  DATETIME NOT NULL,
  PRIMARY KEY (id),
  KEY idx_ping_ts (timesheet_id, recorded_at),
  CONSTRAINT fk_ping_ts FOREIGN KEY (timesheet_id)
    REFERENCES timesheets (id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- Audit log: every manager/admin hour adjustment (who/when/old/new/reason)
CREATE TABLE timesheet_adjustments (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  timesheet_id BIGINT UNSIGNED NOT NULL,
  adjusted_by  BIGINT UNSIGNED NOT NULL,
  old_check_in DATETIME NULL,
  new_check_in DATETIME NULL,
  old_check_out DATETIME NULL,
  new_check_out DATETIME NULL,
  reason       VARCHAR(255) NOT NULL,
  created_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_adj_ts (timesheet_id),
  CONSTRAINT fk_adj_ts   FOREIGN KEY (timesheet_id) REFERENCES timesheets (id) ON DELETE CASCADE,
  CONSTRAINT fk_adj_user FOREIGN KEY (adjusted_by)  REFERENCES users (id)
) ENGINE=InnoDB;

-- ---- Materials & expenses (optional per project) ----------------------------
CREATE TABLE expenses (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  project_id  BIGINT UNSIGNED NOT NULL,
  category    VARCHAR(80) NOT NULL COMMENT 'material, expense, ...',
  vendor_id   BIGINT UNSIGNED NULL,
  description VARCHAR(255) NOT NULL,
  quantity    DECIMAL(10,2) NULL,
  unit_cost   DECIMAL(12,2) NULL,
  amount      DECIMAL(12,2) NOT NULL,
  expense_date DATE NOT NULL,
  receipt_url VARCHAR(500) NULL,
  billed_invoice_id BIGINT UNSIGNED NULL COMMENT 'invoice this material was billed on (NULL = not billed yet)',
  created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_expense_project (project_id),
  KEY idx_expense_billed (billed_invoice_id),
  CONSTRAINT fk_expense_project FOREIGN KEY (project_id) REFERENCES projects (id) ON DELETE CASCADE,
  CONSTRAINT fk_expense_vendor  FOREIGN KEY (vendor_id)  REFERENCES vendors (id) ON DELETE SET NULL
) ENGINE=InnoDB;

-- ---- Invoices ----------------------------------------------------------------
CREATE TABLE invoices (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  client_id    BIGINT UNSIGNED NOT NULL,
  project_id   BIGINT UNSIGNED NULL,
  invoice_no   VARCHAR(40) NOT NULL,
  issue_date   DATE NOT NULL,
  due_date     DATE NULL,
  subtotal     DECIMAL(12,2) NOT NULL DEFAULT 0,
  tax          DECIMAL(12,2) NOT NULL DEFAULT 0,
  total        DECIMAL(12,2) NOT NULL DEFAULT 0,
  amount_paid  DECIMAL(12,2) NOT NULL DEFAULT 0,
  balance      DECIMAL(12,2) NOT NULL DEFAULT 0 COMMENT 'total - amount_paid',
  status       ENUM('open','partial','paid','overdue') NOT NULL DEFAULT 'open',
  notes        TEXT NULL,
  created_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_invoice_no (invoice_no),
  KEY idx_invoice_client (client_id),
  KEY idx_invoice_status (status),
  CONSTRAINT fk_invoice_client  FOREIGN KEY (client_id)  REFERENCES clients (id),
  CONSTRAINT fk_invoice_project FOREIGN KEY (project_id) REFERENCES projects (id) ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE TABLE invoice_items (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  invoice_id  BIGINT UNSIGNED NOT NULL,
  description VARCHAR(255) NOT NULL,
  quantity    DECIMAL(10,2) NOT NULL DEFAULT 1,
  unit_price  DECIMAL(12,2) NOT NULL,
  line_total  DECIMAL(12,2) NOT NULL,
  expense_id  BIGINT UNSIGNED NULL COMMENT 'source material/expense when pulled from expenses',
  PRIMARY KEY (id),
  KEY idx_item_invoice (invoice_id),
  KEY idx_item_expense (expense_id),
  CONSTRAINT fk_item_invoice FOREIGN KEY (invoice_id)
    REFERENCES invoices (id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE invoice_payments (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  invoice_id BIGINT UNSIGNED NOT NULL,
  amount     DECIMAL(12,2) NOT NULL,
  paid_at    DATE NOT NULL,
  method     VARCHAR(60) NULL,
  notes      VARCHAR(255) NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_pay_invoice (invoice_id),
  CONSTRAINT fk_pay_invoice FOREIGN KEY (invoice_id)
    REFERENCES invoices (id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---- Field progress (photos / notes) -----------------------------------------
CREATE TABLE progress_updates (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  project_id BIGINT UNSIGNED NOT NULL,
  job_id     BIGINT UNSIGNED NULL,
  user_id    BIGINT UNSIGNED NOT NULL,
  photo_url  VARCHAR(500) NULL,
  note       TEXT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_prog_project (project_id, created_at),
  CONSTRAINT fk_prog_project FOREIGN KEY (project_id) REFERENCES projects (id) ON DELETE CASCADE,
  CONSTRAINT fk_prog_job     FOREIGN KEY (job_id)     REFERENCES jobs (id) ON DELETE SET NULL,
  CONSTRAINT fk_prog_user    FOREIGN KEY (user_id)    REFERENCES users (id)
) ENGINE=InnoDB;

-- ---- Notifications (in-app; phone push comes with the native app) -------------
CREATE TABLE notifications (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id    BIGINT UNSIGNED NOT NULL COMMENT 'recipient (users.id)',
  type       VARCHAR(32) NOT NULL COMMENT 'out_of_zone, no_show, task_done, task_assigned, timesheet_pending, invoice_overdue, invoice_payment',
  title      VARCHAR(255) NOT NULL,
  detail     TEXT NULL,
  link_view  VARCHAR(24) NOT NULL DEFAULT '' COMMENT 'invoices, projects, entries, payroll, field',
  link_id    BIGINT UNSIGNED NULL COMMENT 'record id inside link_view',
  read_at    DATETIME NULL,
  dedupe_key VARCHAR(120) NOT NULL DEFAULT '' COMMENT 'idempotency key for auto-generated notifications',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_notif_user (user_id, read_at),
  UNIQUE KEY uq_notif_dedupe (user_id, dedupe_key),
  CONSTRAINT fk_notif_user FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---- Payroll payouts (one row per employee per Mon-Sun period; locks it) -----
CREATE TABLE payroll_payouts (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id       BIGINT UNSIGNED NOT NULL,
  period_start  DATE NOT NULL COMMENT 'Monday',
  period_end    DATE NOT NULL COMMENT 'Sunday',
  hours         DECIMAL(8,2) NOT NULL DEFAULT 0,
  regular_hours DECIMAL(8,2) NOT NULL DEFAULT 0,
  ot_hours      DECIMAL(8,2) NOT NULL DEFAULT 0,
  ot_multiplier DECIMAL(4,2) NOT NULL DEFAULT 1,
  days          INT NOT NULL DEFAULT 0,
  gross         DECIMAL(12,2) NOT NULL DEFAULT 0,
  pay_type      VARCHAR(24) NOT NULL DEFAULT 'hora',
  pay_rate      DECIMAL(12,2) NOT NULL DEFAULT 0,
  paid_date     DATE NOT NULL,
  method        VARCHAR(40) NOT NULL DEFAULT 'Check' COMMENT 'Check, Direct deposit, Cash, Other',
  reference     VARCHAR(120) NOT NULL DEFAULT '',
  notes         VARCHAR(255) NOT NULL DEFAULT '',
  created_by    BIGINT UNSIGNED NOT NULL DEFAULT 0,
  created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_payout_period (user_id, period_start, period_end, pay_type, pay_rate),
  CONSTRAINT fk_payout_user FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---- Task work time (timer segments + manual hour logs) -----------------------
CREATE TABLE task_time_logs (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  task_id    BIGINT UNSIGNED NOT NULL,
  user_id    BIGINT UNSIGNED NOT NULL,
  kind       ENUM('timer','manual') NOT NULL DEFAULT 'timer',
  started_at DATETIME NULL,
  ended_at   DATETIME NULL COMMENT 'NULL while the timer is running',
  minutes    DECIMAL(10,2) NOT NULL DEFAULT 0,
  note       VARCHAR(255) NOT NULL DEFAULT '',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_ttl_task (task_id),
  KEY idx_ttl_user (user_id),
  CONSTRAINT fk_ttl_task FOREIGN KEY (task_id) REFERENCES job_tasks (id) ON DELETE CASCADE,
  CONSTRAINT fk_ttl_user FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE
) ENGINE=InnoDB;
